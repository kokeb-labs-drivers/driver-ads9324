57constexpr void Put(RegisterImage& img, reg::Bank bank, uint8_t address, uint16_t value) noexcept {
58 if (address >= reg::kAddressSpace) {

78constexpr void Merge(RegisterImage& img, reg::Bank bank, uint8_t address, uint16_t mask,
79 uint16_t bits) noexcept {
80 const uint32_t kept =
81 static_cast<uint32_t>(Get(img, bank, address)) & ~static_cast<uint32_t>(mask);
82 Put(img, bank, address, static_cast<uint16_t>(kept | (static_cast<uint32_t>(bits) & mask)));

91constexpr void EncodeInterface(RegisterImage& img, const Interface& i) noexcept {
92 using reg::Bank;
93 uint16_t gc3 = reg::DoutLaneSel::Insert(0_u16, Code(i.lanes));
94 gc3 = reg::DoutLength::Insert(gc3, Code(i.length));
95 gc3 = Or(gc3, Bit(i.data_on_sdout, reg::kAdcDataSdoutEn));
96 gc3 = Or(gc3, Bit(i.format == DataFormat::OffsetBinary, reg::kEnOfsBinary));
97 gc3 = Or(gc3, Bit(i.randomize, reg::kEnXorPatt));
98 gc3 = Or(gc3, Bit(HasDiagByte(i), reg::kEnDiagFlag));
99 Put(img, Bank::Common, reg::kGenCfg3, gc3);
100 Put(img, Bank::Common, reg::kXorBitsCtl, i.randomize ? reg::kXorBitsRandomizer : 0_u16);
101 const auto& d = i.delay_ns;
102 uint16_t d1 = reg::DelaySdout::Insert(0_u16, d[0]);
103 d1 = reg::DelayD3::Insert(d1, d[4]);
104 d1 = reg::DelayD2::Insert(d1, d[3]);
105 d1 = reg::DelayD1::Insert(d1, d[2]);
106 d1 = reg::DelayD0::Insert(d1, d[1]);
107 uint16_t d2 = reg::DelayD7::Insert(0_u16, d[8]);
108 d2 = reg::DelayD6::Insert(d2, d[7]);
109 d2 = reg::DelayD5::Insert(d2, d[6]);
110 d2 = reg::DelayD4::Insert(d2, d[5]);
111 Put(img, Bank::Common, reg::kDigDelayCfg1, d1);
112 Put(img, Bank::Common, reg::kDigDelayCfg2, d2);
113 bool any_delay = false;
114 for (const uint8_t v : d) {
115 any_delay = any_delay || (v != 0U);
116 }
117 uint16_t gc4 = reg::DriveStrengthField::Insert(0_u16, Code(i.drive));
118 gc4 = Or(gc4, Bit(any_delay, reg::kDigDelayEn));
119 Put(img, Bank::Common, reg::kGenCfg4, gc4);
120}

122constexpr void EncodeAlarm(RegisterImage& img, const Alarm& a) noexcept {
123 using reg::Bank;
124 const bool or_masked = a.function == AlarmFunction::OrMasked;
125 // Assumption A41: a set mask bit leaves the source out of the OR.
126 const uint32_t mask = or_masked ? (~static_cast<uint32_t>(a.sources) & 0xFFU) : 0U;
127 uint16_t v = reg::DrdyAlrmSel::Insert(0_u16, Code(a.function));
128 v = reg::AlarmMask::Insert(v, mask);
129 v = Or(v, Bit(a.active_low, reg::kAlrmPol));
130 v = Or(v, Bit(a.pulse, reg::kAlrmType));
131 Put(img, Bank::Common, reg::kDrdyAlarmSel, v);
132 // GEN_CFG4 is shared with the interface encoder: merge only ALRM_DIS.
133 Merge(img, Bank::Common, reg::kGenCfg4, reg::kAlrmDis, Bit(a.disabled, reg::kAlrmDis));
134}

166constexpr void EncodeAnalog(RegisterImage& img, const Config& c, uint16_t cal_mode) noexcept {
167 using reg::Bank;
168 Put(img, Bank::Common, reg::kAnaCfg1, ReferenceBits(c.reference));
169 uint16_t a2 = reg::AdcNumSel::Insert(0_u16, Code(c.channels.count));
170 a2 = reg::AdcChSel::Insert(a2, ChannelSelCode(c.channels));
171 a2 = Or(a2, Bit(c.low_speed_crosstalk, reg::kXtalkLowSpeed));
172 Put(img, Bank::Common, reg::kAnaCfg2, a2);
173 uint16_t cal = reg::AdcCalMode::Insert(0_u16, cal_mode);
174 cal = Or(cal, Bit(IsDifferential(c.inputs[0]), reg::kSeDiffAin1To8));
175 cal = Or(cal, Bit(IsDifferential(c.inputs[8]), reg::kSeDiffAin9To16));
176 Put(img, Bank::Common, reg::kAdcCal, cal);
177}

183constexpr void EncodeFilter(RegisterImage& img, const Config& c) noexcept {
184 using reg::Bank;
185 const Filter& f = c.filter;
186 uint16_t v = Bit(AnyPhaseDelay(c), reg::kPhaseDelayEn);
187 if (f.kind == FilterKind::Fir) {
188 v = reg::FirFiltSel::Insert(v, Code(f.fir));
189 } else if (f.kind == FilterKind::MovingAverage) {
190 v = reg::MvgAvgLen::Insert(v, Code(f.length));
191 } else if (f.kind == FilterKind::BlockAverage) {
192 v = reg::BlkAvgOsr::Insert(v, Code(f.length));
193 v = Or(v, Bit(f.internal_clock, reg::kIntTrigMode));
194 } else {
195 v = Or(v, 0_u16);
196 }
197 Put(img, Bank::Common, reg::kDigFilter, v);
198 // One gate for every digital filter (section 7.3.11.2, assumption A31).
199 const bool gate = f.kind != FilterKind::None;
200 Put(img, Bank::Common, reg::kGenCfg5, Bit(gate, Or(reg::kAvgModeOvrEn, reg::kAvgMode)));
201}

218constexpr void EncodeChannel(RegisterImage& img, uint8_t channel, const Input& in) noexcept {
219 const reg::ChannelLocation at = reg::Locate(channel);
220 const uint8_t l = at.local;
221 const uint32_t half = reg::HalfShift(l);
222 const auto pga = static_cast<uint16_t>(static_cast<uint32_t>(PgaByte(in)) << half);
223 Merge(img, at.bank, reg::PgaConfigAddress(l), HalfMask(l, 0x00FF_u16), pga);
224 const uint32_t bw_shift = 2U * static_cast<uint32_t>(l);
225 Merge(img, at.bank, reg::kPgaBwSel, static_cast<uint16_t>(3U << bw_shift),
226 static_cast<uint16_t>(Code(in.bandwidth) << bw_shift));
227 const auto delay = static_cast<uint16_t>(static_cast<uint32_t>(in.phase_delay) << half);
228 Merge(img, at.bank, reg::PhaseDelayAddress(l), HalfMask(l, 0x00FF_u16), delay);
229 Put(img, at.bank, reg::OffsetAddress(l), Twos(in.offset_quarter_lsb, 10U));
230 Put(img, at.bank, reg::GainAddress(l), Twos(in.gain, 14U));
231 const auto th = static_cast<uint16_t>(
232 (static_cast<uint32_t>(static_cast<uint8_t>(in.window.high_code)) << 8U) |
233 static_cast<uint32_t>(static_cast<uint8_t>(in.window.low_code)));
234 Put(img, at.bank, reg::DwcThAddress(l), th);
235 const auto hys = static_cast<uint16_t>(static_cast<uint32_t>(in.window.hysteresis) << half);
236 Merge(img, at.bank, reg::DwcHysAddress(l), HalfMask(l, 0x00FF_u16), hys);
237 const auto en = static_cast<uint16_t>(in.window.enabled ? (1U << l) : 0U);
238 Merge(img, at.bank, reg::kDwcCfg, static_cast<uint16_t>(1U << l), en);
239}

241constexpr void EncodeBankSettings(RegisterImage& img, const Config& c) noexcept {
242 for (uint32_t g = 0U; g < 2U; ++g) {
243 const reg::Bank bank = (g == 0U) ? reg::Bank::Ain1To8 : reg::Bank::Ain9To16;
244 Merge(img, bank, reg::kDwcCfg, 0x0F00_u16,
245 reg::DwcGlitchFilt::Insert(0_u16, c.window_glitch[g]));
246 uint16_t corr = Bit(!c.correction.offset_enabled, reg::kOfsCorrDis);
247 corr = Or(corr, Bit(!c.correction.gain_enabled, reg::kGanCorrDis));
248 Put(img, bank, reg::kCorrection, corr);
249 const TestPattern& t = c.test_pattern[g];
250 uint16_t cfg = reg::TpMode::Insert(0_u16, Code(t.kind));
251 cfg = Or(cfg, Bit(t.enabled, reg::kTpEn));
252 cfg = Or(cfg, Bit(!t.index_insertion, reg::kTpDisIdx));
253 cfg = Or(cfg, Bit(t.step_every_convst, reg::kTpUpdMode));
254 Put(img, bank, reg::kTpCfg, cfg);
255 for (uint32_t i = 0U; i < 8U; ++i) {
256 const reg::ChannelLocation at = reg::Locate(static_cast<uint8_t>((8U * g) + i));
257 Put(img, at.bank, reg::TpAddress(at.local), t.value[i]);
258 }
259 }

269constexpr RegisterImage Encode(const Config& c, uint16_t cal_mode) noexcept {
270 RegisterImage img{};
271 EncodeInterface(img, c.interface);
272 EncodeAlarm(img, c.alarm);
273 EncodeAnalog(img, c, cal_mode);
274 EncodeFilter(img, c);
275 for (uint8_t ch = 0U; ch < reg::kChannelCount; ++ch) {
276 EncodeChannel(img, ch, c.inputs[ch]);
277 }
278 EncodeBankSettings(img, c);
279 return img;
280}

316constexpr void DecodeWords(const Config& c, std::span<const uint8_t> bytes, uint16_t mask,
317 Frame& frame) noexcept {
318 const uint32_t per = (c.interface.length == WordLength::Bits24) ? 3U : 2U;
319 const uint32_t n = FrameChannels(c.channels);
320 for (uint32_t slot = 0U; slot < n; ++slot) {
321 const uint32_t at = slot * per;
322 const uint8_t ch = FrameSlotChannel(c.channels, slot);
323 const uint32_t word =
324 (static_cast<uint32_t>(bytes[at]) << 8U) | static_cast<uint32_t>(bytes[at + 1U]);
325 const auto raw = static_cast<uint16_t>(word ^ static_cast<uint32_t>(mask));
326 const int16_t code = WireToCode(raw, c.interface.format);
327 frame.raw[ch] = raw;
328 frame.codes[ch] = code;
329 frame.low_byte[ch] = (per == 3U) ? bytes[at + 2U] : static_cast<uint8_t>(0U);
330 frame.microvolts[ch] = CodeToMicrovolts(code, c.inputs[ch].range);
331 frame.present_mask =
332 static_cast<uint16_t>(static_cast<uint32_t>(frame.present_mask) | (1U << ch));
333 if (IsClipped(code)) {
334 frame.clipped_mask =
335 static_cast<uint16_t>(static_cast<uint32_t>(frame.clipped_mask) | (1U << ch));
336 }
337 }
338}

347constexpr Result<Frame> DecodeFrame(const Config& c, std::span<const uint8_t> bytes) noexcept {
348 if (bytes.size() != FrameBytes(c.interface, c.channels)) {
349 return Result<Frame>::Failure(Error::InvalidArgument);
350 }
351 Frame frame{};
352 const uint32_t per = (c.interface.length == WordLength::Bits24) ? 3U : 2U;
353 const uint32_t trailer = FrameChannels(c.channels) * per;
354 const bool diag = HasDiagByte(c.interface);
355 frame.diag = diag ? bytes[trailer] : static_cast<uint8_t>(0U);
356 frame.prbs = c.interface.randomize ? bytes[trailer + 1U] : static_cast<uint8_t>(0U);
357 const uint16_t mask =
358 c.interface.randomize ? DerandomizeMask(frame.prbs) : static_cast<uint16_t>(0U);
359 DecodeWords(c, bytes, mask, frame);
360 frame.window_alert = (static_cast<uint32_t>(frame.diag) & kDiagWindowAlert) != 0U;
361 frame.reset_detected = (static_cast<uint32_t>(frame.diag) & kDiagResetDetect) != 0U;
362 return Result<Frame>::Success(frame);
363}
