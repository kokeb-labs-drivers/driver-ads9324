36// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept
37// the exact signature, `noexcept` and the result type; the driver rejects a
38// hook that is declared but does not match (KB-CPP20-007).

134constexpr uint32_t FlushPhase(reg::Bank bank, uint8_t address) noexcept {
135 if ((bank == reg::Bank::Common) && (address == reg::kGenCfg5)) {
136 return 0U;
137 }
138 return ((bank != reg::Bank::Common) && (address == reg::kDwcCfg)) ? 2U : 1U;
139}
140
143constexpr bool IsFilterRegister(reg::Bank bank, uint8_t address) noexcept {
144 if (bank == reg::Bank::Common) {
145 return (address == reg::kDigFilter) || (address == reg::kGenCfg5);
146 }
147 return (address >= reg::kPhaseDelay) && (address < (reg::kPhaseDelay + 4U));
148}
149
150static_assert(FlushPhase(reg::Bank::Common, reg::kGenCfg5) == 0U);
151static_assert(FlushPhase(reg::Bank::Ain9To16, reg::kDwcCfg) == 2U);
152static_assert(FlushPhase(reg::Bank::Ain1To8, reg::kDwcTh) == 1U);
153static_assert(IsFilterRegister(reg::Bank::Ain9To16, reg::kPhaseDelay + 3U));
154static_assert(!IsFilterRegister(reg::Bank::Ain9To16, reg::kOffset));
