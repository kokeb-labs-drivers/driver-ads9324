740constexpr Error ValidateInterface(const Interface& i) noexcept {
741 bool ok =
742 IsEnumerator(i.lanes) && IsEnumerator(i.format) && IsEnumerator(i.drive) && IsValid(i.length);
743 // One data lane on SDOUT is the only lane layout that uses SDOUT (Table
744 // 7-11).
745 ok = ok && (!i.data_on_sdout || (i.lanes == Lanes::One));
746 // The randomizer is not specified for 24-bit words (section 7.3.14.4).
747 ok = ok && (!i.randomize || (i.length == WordLength::Bits16));
748 for (const uint8_t d : i.delay_ns) {
749 ok = ok && (d <= kDelayMaxNs);
750 }
751 return Check(ok);
752}

795constexpr Error ValidateFilter(const Filter& f) noexcept {
796 bool ok = IsEnumerator(f.kind) && IsEnumerator(f.length) && IsValid(f.fir);
797 const bool average =
798 (f.kind == FilterKind::BlockAverage) || (f.kind == FilterKind::MovingAverage);
799 ok = ok && (!average || (f.length != AverageLength::None));
800 ok = ok && (!f.internal_clock || (f.kind == FilterKind::BlockAverage));

804constexpr Error ValidateAlarm(const Alarm& a) noexcept {
805 bool ok = IsValid(a.function);
806 const bool or_masked = a.function == AlarmFunction::OrMasked;

808 ok = ok && (!or_masked || (a.or_mask_acknowledged && (a.sources != 0U)));
809 ok = ok && (!a.pulse || (a.function == AlarmFunction::WindowAlert));

828constexpr Error ValidateExtras(const Config& c) noexcept {

861constexpr Error Validate(const Config& c) noexcept {
862 if (c.daisy.enabled) {
863 return Error::Unsupported;
864 }
865 const std::array<Error, 8> steps{detail::ValidateInterface(c.interface),
866 detail::ValidateInputs(c.inputs),
867 detail::ValidateFilter(c.filter),
868 detail::ValidateAlarm(c.alarm),
869 detail::ValidateSampling(c.sampling),
870 detail::ValidateSupply(c.supply),
871 detail::ValidateExtras(c),
872 detail::Check(!detail::WindowWithOffsetBinary(c))};
873 for (const Error e : steps) {
874 if (e != Error::Ok) {
875 return e;
876 }
877 }
878 return Error::Ok;
879}

1047constexpr int16_t OffsetCorrectionFor(int32_t expected_code, int32_t measured_code) noexcept {

1061constexpr int16_t GainCorrectionFor(int32_t expected_code, int32_t measured_code) noexcept {
1062 if (measured_code == 0) {
1063 return 0;
1064 }
1065 const int64_t num = static_cast<int64_t>(expected_code - measured_code) * 65536;
1066 const int64_t den = measured_code;
1067 const int64_t half = ((den < 0) ? -den : den) / 2;
1068 const bool same_sign = (num >= 0) == (den > 0);
1069 const int64_t v = same_sign ? ((((num < 0) ? -num : num) + half) / ((den < 0) ? -den : den))
1070 : -((((num < 0) ? -num : num) + half) / ((den < 0) ? -den : den));
1071 const int64_t clamped = std::clamp<int64_t>(v, kGainMin, kGainMax);
1072 return static_cast<int16_t>(clamped);
1073}

1179constexpr uint8_t FrameSlotChannel(const ChannelSelect& select, uint32_t slot) noexcept {
1180 const auto group = static_cast<uint32_t>(select.group);
1181 uint32_t channel = slot;
1182 switch (select.count) {
1183 case ChannelCount::Eight:
1184 if (group != 0U) {
1185 channel = slot + 4U;
1186 } else if (slot >= 4U) {
1187 channel = slot + 8U;
1188 }
1189 break;
1190 case ChannelCount::Four:
1191 channel = (slot < 2U) ? (slot + (2U * group)) : (slot + 12U - (2U * group));
1192 break;
1193 case ChannelCount::Two:
1194 channel = (slot == 0U) ? group : (15U - group);
1195 break;
1196 case ChannelCount::All16:
1197 default:
1198 break;
1199 }
1200 return static_cast<uint8_t>(channel);
