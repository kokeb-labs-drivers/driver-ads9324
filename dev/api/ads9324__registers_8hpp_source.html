365constexpr bool IsChannel(uint8_t address) noexcept {
366 // PGA_CONFIG (0x08) to TP_AIN8 (0x36) is one run of registers, then GEN_CFG5
367 // and DWC_FLAG.
368 return ((address >= kPgaConfig) && (address < kTp + 8U)) || (address == kCorrection) ||
369 (address == kDwcFlag);
370}

382constexpr bool IsDocumented(Bank bank, uint8_t address) noexcept {
383 return (bank == Bank::Common) ? detail::IsCommon(address) : detail::IsChannel(address);

393constexpr bool IsReadOnly(Bank bank, uint8_t address) noexcept {
394 if (bank == Bank::Common) {
395 return (address == kDeviceStatus) || (address == kDeviceId);
396 }
397 return address == kDwcFlag;
398}

408constexpr bool IsShadowed(Bank bank, uint8_t address) noexcept {
409 if (!IsDocumented(bank, address) || IsReadOnly(bank, address)) {
410 return false;
411 }
412 const bool command_register =
413 (bank == Bank::Common) && ((address == kGenCfg1) || (address == kBankSel));
414 return !command_register;
415}

424constexpr uint16_t ResetValue(Bank bank, uint8_t address) noexcept {
425 if (bank == Bank::Common) {
426 if (address == kDeviceId) {
427 return kDeviceIdValue;
428 }
429 return (address == kBankSel) ? 1_u16 : 0_u16;
430 }
431 const bool threshold = (address >= kDwcTh) && (address < kDwcTh + 8U);
432 const bool hysteresis = (address > kDwcHys) && (address < kDwcHys + 4U);
433 return (threshold || hysteresis) ? 0xFF00_u16 : 0_u16;
434}

440constexpr uint32_t CountDocumented() noexcept {
441 uint32_t count = 0U;
442 for (uint32_t b = 0U; b < kBankCount; ++b) {
443 for (uint32_t a = 0U; a < kAddressSpace; ++a) {
444 if (IsDocumented(static_cast<Bank>(b), static_cast<uint8_t>(a))) {
445 ++count;
446 }
447 }
448 }
449 return count;
450}

452// The register map is the datasheet's: 18 common registers and 49 in each
453// channel bank.
454static_assert(CountDocumented() == 116U);
455static_assert(IsDocumented(Bank::Common, kDeviceId) && !IsDocumented(Bank::Common, 0x00));
456static_assert(IsDocumented(Bank::Ain1To8, kDwcFlag) && !IsDocumented(Bank::Ain1To8, 0x38));
457static_assert(!IsDocumented(Bank::Ain9To16, 0x13 + 0x40));
458// Reset values (Tables 8-1 to 8-71).
459static_assert(ResetValue(Bank::Common, kBankSel) == 1U);
460static_assert(ResetValue(Bank::Common, kDeviceId) == 4U);
461static_assert(ResetValue(Bank::Ain1To8, kDwcTh) == 0xFF00U);
462static_assert(ResetValue(Bank::Ain9To16, kDwcHys) == 0U);
463static_assert(ResetValue(Bank::Ain9To16, kDwcHys + 1U) == 0xFF00U);
464// The reversed bank (section 8.3): AIN16 is local 0, AIN9 is local 7; both
465// banks round trip.
466static_assert(Locate(0).bank == Bank::Ain1To8 && Locate(0).local == 0U);
467static_assert(Locate(7).bank == Bank::Ain1To8 && Locate(7).local == 7U);
468static_assert(Locate(15).bank == Bank::Ain9To16 && Locate(15).local == 0U);
469static_assert(Locate(8).bank == Bank::Ain9To16 && Locate(8).local == 7U);
470static_assert(ChannelAt(Locate(11).bank, Locate(11).local) == 11U);
471// AIN15 (channel 14) is local 1: the high half of PGA_CONFIG at 0x08 (Table
472// 8-74).
473static_assert(PgaConfigAddress(Locate(14).local) == 0x08U && HalfShift(Locate(14).local) == 8U);
474// AIN9 (channel 8) is local 7: OFS at 0x18, TP at 0x36 (Tables 8-72).
475static_assert(OffsetAddress(Locate(8).local) == 0x18U && TpAddress(Locate(8).local) == 0x36U);
476static_assert(DwcHysAddress(Locate(8).local) == 0x2DU);
