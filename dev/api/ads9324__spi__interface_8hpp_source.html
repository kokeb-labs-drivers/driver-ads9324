158 Error Write(uint8_t address, uint16_t value) noexcept {
159 tx_[0] = address;
160 tx_[1] = static_cast<uint8_t>(value >> 8U);
161 tx_[2] = static_cast<uint8_t>(value & 0xFFU);
162 return port_.Transfer(tx_.data(), nullptr, kFrameBytes) ? Error::Ok : Error::Bus;
163 }
164
169 Result<uint16_t> Read(uint8_t address) noexcept {
170 if (const Error e = Write(reg::kGenCfg1, ReadCommand(address)); e != Error::Ok) {
171 return Result<uint16_t>::Failure(e);
172 }
173 tx_.fill(0U);
174 rx_.fill(0U);
175 if (!port_.Transfer(tx_.data(), rx_.data(), kFrameBytes)) {
176 return Result<uint16_t>::Failure(Error::Bus);
177 }
178 if (rx_[2] != 0U) {
179 return Result<uint16_t>::Failure(Error::Bus);
180 }
181 return Result<uint16_t>::Success(static_cast<uint16_t>((static_cast<uint32_t>(rx_[0]) << 8U) |
182 static_cast<uint32_t>(rx_[1])));
183 }
184
188 Error ReadStream(std::span<uint8_t> out) noexcept {
189 if (out.empty() || (out.size() > kMaxFrameBytes)) {
190 return Error::InvalidArgument;
191 }
192 zeros_.fill(0U);
193 return port_.Transfer(zeros_.data(), out.data(), out.size()) ? Error::Ok : Error::Bus;
194 }
195
196private:
197 Port& port_;
198 // The transfer buffers belong to the driver object, never to a caller's stack, so a port whose
199 // bus reads them by DMA reaches them (KB-HW-019).
200 std::array<uint8_t, kFrameBytes> tx_{};
201 std::array<uint8_t, kFrameBytes> rx_{};
202 std::array<uint8_t, kMaxFrameBytes> zeros_{};
203};
